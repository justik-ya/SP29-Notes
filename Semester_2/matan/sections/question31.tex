\section{Равномерная сходимость ряда в круге сходимости.}

\begin{Theorem}{Равномерная сходимость и непрерывность суммы}{}
	Пусть степенной ряд $\sum_{n=0}^{\infty} a_n z^n$ имеет радиус сходимости $R > 0$ и пусть $0 < r < R$. Тогда ряд сходится абсолютно и равномерно в замкнутом круге $\{|z| \le r\}$, а его сумма $S(z)$ равномерно непрерывна на $\{|z| \le r\}$.
\end{Theorem}

\begin{proof}
	\textbf{Равномерная сходимость.} Выберем промежуточное $r_1$ с $r < r_1 < R$. Так как $r_1 < R$, ряд $\sum |a_n| r_1^n$ сходится, поэтому его члены ограничены: существует $C$ с $|a_n| r_1^n \le C$ для всех $n$. Тогда при $|z| \le r$ и любом $m \ge 1$
	\[
	\Bigl| \sum_{k=n+1}^{n+m} a_k z^k \Bigr|
	\le \sum_{k=n+1}^{n+m} |a_k|\,|z|^k
	\le \sum_{k=n+1}^{n+m} |a_k| r_1^k \Bigl(\frac{r}{r_1}\Bigr)^k
	\le C \sum_{k=n+1}^{n+m} \Bigl(\frac{r}{r_1}\Bigr)^k.
	\]
	Поскольку $0 < r/r_1 < 1$, хвост геометрического ряда $\sum (r/r_1)^k$ стремится к нулю при $n \to \infty$ равномерно по $m$ и по $z$ (оценка от $z$ не зависит). По критерию Коши для равномерной сходимости ряд сходится абсолютно и равномерно на $\{|z| \le r\}$.

	\textbf{Равномерная непрерывность суммы.} Обозначим $S_n(z) = \sum_{k=0}^{n} a_k z^k$. Зафиксируем $\eps > 0$. По равномерной сходимости найдётся $N_0$ такое, что
	\[
	\sup_{|z| \le r} |S_{N_0}(z) - S(z)| < \frac{\eps}{3}.
	\]
	Многочлен $S_{N_0}$ непрерывен на компакте $\{|z| \le r\}$, а значит, равномерно непрерывен на нём: существует $\delta > 0$ такое, что для всех $z_1, z_2$ с $|z_1|, |z_2| \le r$ и $|z_1 - z_2| < \delta$
	\[
	|S_{N_0}(z_1) - S_{N_0}(z_2)| < \frac{\eps}{3}.
	\]
	Тогда по неравенству треугольника
	\[
	|S(z_1) - S(z_2)| \le |S(z_1) - S_{N_0}(z_1)| + |S_{N_0}(z_1) - S_{N_0}(z_2)| + |S_{N_0}(z_2) - S(z_2)| < \frac{\eps}{3} + \frac{\eps}{3} + \frac{\eps}{3} = \eps.
	\]
	Значит, $S$ равномерно непрерывна на $\{|z| \le r\}$.
\end{proof}

\begin{Corollary}{Непрерывность суммы в круге сходимости}{}
	Сумма степенного ряда $S(z) = \sum_{n=0}^{\infty} a_n z^n$ непрерывна в открытом круге сходимости $\{|z| < R\}$.
\end{Corollary}

\begin{proof}
	Для любой точки $z_*$ с $|z_*| < R$ выберем $r$ так, что $|z_*| < r < R$. По теореме $S$ равномерно непрерывна на $\{|z| \le r\}$, в частности непрерывна в точке $z_*$. Так как $z_*$ произвольна, $S$ непрерывна во всём круге $\{|z| < R\}$.
\end{proof}

\begin{Remark}{Сходимость не обязана быть равномерной во всём круге}{}
	Равномерность гарантируется лишь на замкнутых кругах $\{|z| \le r\}$ строго меньшего радиуса $r < R$. На всём открытом круге $\{|z| < R\}$ сходимость, вообще говоря, лишь поточечная: например, для $\sum z^n$ (где $R = 1$) при $z \to 1^-$ остаток не стремится к нулю равномерно.
\end{Remark}
